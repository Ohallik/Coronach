        [Serializable]
        public sealed class GeneratedStageFile
        {
            public string source;
            public string destination;
        }

        /// <summary>
        /// Optional exact-file form of unity-stage.json. Legacy two-field markers keep
        /// their existing Generated&lt;Batch&gt; behavior. P73 uses this reviewed form so
        /// an opaque cache name cannot choose an arbitrary Assets path and an
        /// intermediate FBX cannot enter Unity by accident.
        /// </summary>
        [Serializable]
        public sealed class GeneratedStageManifest
        {
            public string batch;
            public string gate;
            public string prompt_id;
            public string destinationPack;
            public string destinationSubpath;
            public GeneratedStageFile[] files;
        }

        public static void StagePacks()
        {
            try
            {
                int total = 0;
                foreach (var (pack, sub) in Stage)
                    total += StageOne(pack, sub);
                foreach (var (pack, sub, pattern) in StageFiltered)
                    total += StageOne(pack, sub, pattern);
                foreach (var (pack, sub, dest) in StageAs)
                    total += StageOne(pack, sub, destName: dest);
                total += StageGenerated();
                AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);
                Debug.Log($"[Lattice] PACKS_STAGED files={total}");
                EditorApplication.Exit(0);
            }
            catch (Exception e)
            {
                Debug.LogError($"[Lattice] StagePacks FAILED: {e}");
                EditorApplication.Exit(1);
            }
        }

        internal static int StageOne(string pack, string sub, string pattern = "*", string destName = null)
        {
            var srcRoot = Path.GetFullPath(Path.Combine(ArtSrc, pack, sub ?? ""));
            // Single-file staging (curated-subset law: one model from a 40-model pack).
            if (File.Exists(srcRoot))
            {
                var fileDest = Path.Combine(PacksRoot, pack, destName ?? Path.GetFileName(srcRoot));
                Directory.CreateDirectory(Path.GetDirectoryName(fileDest)!);
                if (!File.Exists(fileDest) || new FileInfo(fileDest).Length != new FileInfo(srcRoot).Length)
                {
                    File.Copy(srcRoot, fileDest, overwrite: true);
                    Debug.Log($"[Lattice] staged {pack}/{sub}: 1 file");
                    return 1;
                }
                return 0;
            }
            if (!Directory.Exists(srcRoot))
            {
                Debug.LogWarning($"[Lattice] pack source missing: {srcRoot}");
                return 0;
            }

            // Texture dirs keep a "Textures" parent so Unity's FBX material import can
            // resolve external textures by its standard sibling-folder search.
            var destPrefix = Path.GetFileName(srcRoot.TrimEnd('\\', '/')).Equals("Textures", StringComparison.OrdinalIgnoreCase)
                ? "Textures" : "";

            int copied = 0;
            bool generated = pack.StartsWith("Generated", StringComparison.Ordinal);
            foreach (var file in Directory.EnumerateFiles(srcRoot, pattern, SearchOption.AllDirectories))
            {
                if (!Extensions.Contains(Path.GetExtension(file).ToLowerInvariant()))
                    continue;
                if (generated && IsGeneratedIntermediate(file))
                    continue;
                var rel = Path.GetRelativePath(srcRoot, file);
                var dest = Path.Combine(PacksRoot, pack, destPrefix, rel);
                Directory.CreateDirectory(Path.GetDirectoryName(dest)!);
                if (!File.Exists(dest) || new FileInfo(dest).Length != new FileInfo(file).Length)
                {
                    File.Copy(file, dest, overwrite: true);
                    copied++;
                }
            }
            Debug.Log($"[Lattice] staged {pack}/{sub}: {copied} files");
            return copied;
        }

        /// <summary>
        /// P30 generative-art gate. Meshy/Blender outputs cache under
        /// art-src/Generated/&lt;Batch&gt;/ and may enter Unity only here, becoming the
        /// isolated pack Assets/_Project/Art/Packs/Generated&lt;Batch&gt;/. The source
        /// directory name is sanitized so an external batch can never choose an
        /// arbitrary Assets destination.
        ///
        /// Generated files that are PIPELINE INTERMEDIATES, not shipped assets.
        ///
        /// The curated-subset law applies to generated content too, and more sharply:
        /// a Meshy character caches four FBX variants and two 2048 albedos, of which
        /// exactly one mesh and one texture ever reach the game. Staging the whole
        /// batch put 981 MB into Assets — against a 1.3 GB repository — for files
        /// nothing references. art-src keeps them (it is git-ignored and it is the
        /// provenance cache); Assets gets what ships.
        ///
        ///   *_Meshy.fbx            the un-rigged generation output; the rigged
        ///                          derivative is what every prefab points at
        ///   *_walking/_running     Meshy's own takes, superseded by the UAL bake
        ///   *_reference.png        the input photo, not a texture
        /// </summary>
        static bool IsGeneratedIntermediate(string file)
        {
            var name = Path.GetFileNameWithoutExtension(file);
            return name.EndsWith("_Meshy", StringComparison.OrdinalIgnoreCase)
                || name.EndsWith("_Meshy_Rigged", StringComparison.OrdinalIgnoreCase)
                || name.EndsWith("_walking_fbx", StringComparison.OrdinalIgnoreCase)
                || name.EndsWith("_running_fbx", StringComparison.OrdinalIgnoreCase)
                || name.EndsWith("_walking", StringComparison.OrdinalIgnoreCase)
                || name.EndsWith("_running", StringComparison.OrdinalIgnoreCase)
                || name.EndsWith("_reference", StringComparison.OrdinalIgnoreCase);
        }

        static int StageGenerated()
        {
            var root = Path.Combine(ArtSrc, "Generated");
            if (!Directory.Exists(root)) return 0;
            int total = 0;
            foreach (var dir in Directory.EnumerateDirectories(root).OrderBy(p => p, StringComparer.Ordinal))
            {
                // A generated cache is not automatically an import request. The
                // Meshy bridge writes this marker for batches that intentionally
                // enter the quarantine/style gate; older caches remain untouched.
                if (!File.Exists(Path.Combine(dir, "unity-stage.json"))) continue;
                var batch = Path.GetFileName(dir);
                var safe = new string(batch.Where(c => char.IsLetterOrDigit(c) || c == '-' || c == '_').ToArray());
                if (string.IsNullOrWhiteSpace(safe) || !safe.Equals(batch, StringComparison.Ordinal))
                    throw new InvalidOperationException($"Unsafe generated batch name: {batch}");
                total += StageGeneratedBatch(dir, batch, safe);
            }
            return total;
        }

        static int StageGeneratedBatch(string sourceDirectory, string batch, string safeBatch)
        {
            string marker = Path.Combine(sourceDirectory, "unity-stage.json");
            string json = File.ReadAllText(marker);
            var manifest = JsonUtility.FromJson<GeneratedStageManifest>(json);
            if (manifest == null)
                throw new InvalidOperationException(
                    $"Generated stage marker did not produce an object: '{marker}'");

            // The legacy marker has no destination fields and intentionally stages the
            // curated batch directory. Once any exact-mapping field is present, however,
            // a missing/empty files array is a broken exact manifest -- never permission
            // to fall back to broad staging.
            bool exactMapping = manifest.files != null
                || !string.IsNullOrWhiteSpace(manifest.destinationPack)
                || !string.IsNullOrWhiteSpace(manifest.destinationSubpath);
            if (!exactMapping)
                return StageOne("Generated" + safeBatch,
                    Path.Combine("..", "Generated", batch));

            string[] issues = ValidateGeneratedStageManifest(manifest, batch);
            if (issues.Length != 0)
                throw new InvalidOperationException($"Unsafe generated stage manifest '{marker}': "
                                                    + string.Join("; ", issues));

            string destinationRoot = Path.Combine(PacksRoot, manifest.destinationPack,
                manifest.destinationSubpath ?? string.Empty);
            int copied = 0;
            foreach (GeneratedStageFile file in manifest.files)
            {
                string source = SafeChildPath(sourceDirectory, file.source);
                if (!File.Exists(source))
                    throw new FileNotFoundException(
                        $"Generated stage source missing for {batch}: {file.source}", source);
                string destination = SafeChildPath(destinationRoot, file.destination);
                Directory.CreateDirectory(Path.GetDirectoryName(destination)!);
                if (!FilesEqual(source, destination))
                {
                    File.Copy(source, destination, overwrite: true);
                    copied++;
                }
            }
            Debug.Log($"[Lattice] staged Generated{batch}/manifest: {copied} files");
            return copied;
        }

        public static string[] ValidateGeneratedStageManifestJson(string json, string batch)
        {
            if (string.IsNullOrWhiteSpace(json))
                return new[] { "manifest JSON is empty" };
            try
            {
                return ValidateGeneratedStageManifest(
                    JsonUtility.FromJson<GeneratedStageManifest>(json), batch);
            }
            catch (Exception exception)
            {
                return new[] { "manifest JSON is invalid: " + exception.Message };
            }
        }

        static string[] ValidateGeneratedStageManifest(GeneratedStageManifest manifest,
            string batch)
        {
            var issues = new List<string>();
            if (manifest == null)
                return new[] { "manifest JSON did not produce an object" };
            if (!string.Equals(manifest.batch, batch, StringComparison.Ordinal))
                issues.Add($"batch '{manifest.batch}' != source directory '{batch}'");
            if (!string.Equals(manifest.gate, "PackStaging", StringComparison.Ordinal))
                issues.Add($"gate '{manifest.gate}' != PackStaging");
            if (string.IsNullOrWhiteSpace(manifest.prompt_id))
                issues.Add("prompt_id is required for an exact generated mapping");
            if (!IsSafeSegment(manifest.destinationPack)
                || !manifest.destinationPack.StartsWith("Generated", StringComparison.Ordinal))
                issues.Add($"unsafe destinationPack '{manifest.destinationPack}'");
            if (!IsSafeRelativePath(manifest.destinationSubpath, allowEmpty: true))
                issues.Add($"unsafe destinationSubpath '{manifest.destinationSubpath}'");
            if (manifest.files == null || manifest.files.Length == 0)
                issues.Add("files must contain at least one exact mapping");
            else
            {
                var destinations = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
                foreach (GeneratedStageFile file in manifest.files)
                {
                    if (file == null || !IsSafeRelativePath(file.source, allowEmpty: false))
                    {
                        issues.Add($"unsafe source '{file?.source}'");
                        continue;
                    }
                    if (!IsSafeRelativePath(file.destination, allowEmpty: false))
                    {
                        issues.Add($"unsafe destination '{file.destination}'");
                        continue;
                    }
                    string sourceExtension = Path.GetExtension(file.source).ToLowerInvariant();
                    string destinationExtension = Path.GetExtension(file.destination).ToLowerInvariant();
                    if (!Extensions.Contains(sourceExtension) || sourceExtension != destinationExtension)
                        issues.Add($"mapping extension mismatch '{file.source}' -> '{file.destination}'");
                    if (!destinations.Add(file.destination.Replace('\\', '/')))
                        issues.Add($"duplicate destination '{file.destination}'");
                }
            }
            return issues.ToArray();
        }

        static bool IsSafeSegment(string value) => !string.IsNullOrWhiteSpace(value)
            && value.All(c => char.IsLetterOrDigit(c) || c == '-' || c == '_' || c == '.');

        static bool IsSafeRelativePath(string value, bool allowEmpty)
        {
            if (string.IsNullOrWhiteSpace(value))
                return allowEmpty;
            if (Path.IsPathRooted(value))
                return false;
            string[] segments = value.Replace('\\', '/').Split('/');
            return segments.Length > 0 && segments.All(segment =>
                segment != "." && segment != ".." && IsSafeSegment(segment));
        }

        static string SafeChildPath(string root, string relative)
        {
            string absoluteRoot = Path.GetFullPath(root).TrimEnd(Path.DirectorySeparatorChar,
                Path.AltDirectorySeparatorChar) + Path.DirectorySeparatorChar;
            string candidate = Path.GetFullPath(Path.Combine(absoluteRoot, relative));
            if (!candidate.StartsWith(absoluteRoot, StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException($"Path escapes staging root: {relative}");
            return candidate;
        }

        static bool FilesEqual(string source, string destination)
        {
            if (!File.Exists(destination))
                return false;
            var sourceInfo = new FileInfo(source);
            var destinationInfo = new FileInfo(destination);
            if (sourceInfo.Length != destinationInfo.Length)
                return false;
            const int bufferSize = 1024 * 128;
            var leftBuffer = new byte[bufferSize];
            var rightBuffer = new byte[bufferSize];
            using var left = File.OpenRead(source);
            using var right = File.OpenRead(destination);
            while (true)
            {
                int leftRead = left.Read(leftBuffer, 0, leftBuffer.Length);
                int rightRead = right.Read(rightBuffer, 0, rightBuffer.Length);
                if (leftRead != rightRead)
                    return false;
                if (leftRead == 0)
                    return true;
                for (int index = 0; index < leftRead; index++)
                    if (leftBuffer[index] != rightBuffer[index])
                        return false;
            }
        }
    }
